●Declare DAO as an interface or abstract class.
●Room creates DAO implementation at compile time.
●Room verifies all of your DAO queries at compile-time.
It is “Data access”, not limited to “Database access”

fun getInstance(context: Context): ColorDatabase {
    return INSTANCE ?: synchronized(this) {
        INSTANCE ?: Room.databaseBuilder(
            context.applicationContext,
            ColorDatabase::class.java, "color_database"
        )
        .fallbackToDestructiveMigration()
        .build()
        .also { INSTANCE = it }
    }
}
public abstract class ColorDatabase extends RoomDatabase {
  private static volatile ColorDatabase INSTANCE;
  public static ColorDatabase getInstance(Context context) {
    if (INSTANCE == null) {
      synchronized (ColorDatabase.class) {
        if (INSTANCE == null) {
          INSTANCE = Room.databaseBuilder(
              context.getApplicationContext(),
              ColorDatabase.class,
              "color_database"
          )
          .fallbackToDestructiveMigration()
          .build();
        }
      }
    }
    return INSTANCE;
  }
}
